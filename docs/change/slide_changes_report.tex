\section*{Detailed Breakdown of New Slides}

\subsection*{1. Slide 6: Stage-by-Stage Pipeline Processing (Inserted after Slide 5)}
\begin{itemize}
    \item \textbf{Stage 1 (Indexing):} Metadata enrichment (\texttt{Title $\parallel$ Header $\parallel$ Passage}) + \texttt{underthesea} BM25 term tokenization.
    \item \textbf{Stage 2 (KG Overlay):} Dynamic \texttt{id\_str\_filter} validity masking + 1-hop \texttt{CHUNK\_NEXT} reading-order expansion.
    \item \textbf{Stage 3 (Fusion \& Rerank):} RRF ($k=60$) candidate fusion ($K=30$) + GPU load-run-unload Cross-Encoder reranking ($N=10$).
    \item \textbf{Stage 4 (Generation):} \texttt{gpt-4o-mini} ($T=0.0$) with statutory citation grounding and \texttt{INSUFFICIENT\_CONTEXT} abstention.
\end{itemize}

\subsection*{2. Slide 7: Knowledge Graph Schema and Construction (Inserted after Slide 6)}
\begin{itemize}
    \item \textbf{Graph Node Hierarchy:} Formalizes Document ($D$), Provision ($P$), and Chunk ($C$) entity nodes.
    \item \textbf{Edge Schema:} Distinguishes Structural edges (\texttt{DOC\_HAS\_PROVISION}, \texttt{PROVISION\_HAS\_CHUNK}, \texttt{CHUNK\_NEXT}, \texttt{PROVISION\_NEXT}) from Legal Relations (amend, replace, cite, repeal).
    \item \textbf{Ingested Scale Table:} Embeds the formal corpus statistics table (151,624 Documents, 1,386,267 Provisions, 1,513,376 Chunks, and 883,256 Document Edges).
    \item \textbf{Validity Rule:} Highlights dynamic \texttt{as\_of\_date} whitelisting and quarantining of unverified records.
\end{itemize}
