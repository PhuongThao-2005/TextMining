\begin{frame}{Knowledge Graph Schema}
\small
\begin{columns}[T]
  \column{0.42\textwidth}
    \textbf{Node types}
    \begin{itemize}
      \item \textbf{Document}: authority, issuer, effective date, validity.
      \item \textbf{Provision}: article/clause header and unit type.
      \item \textbf{Chunk}: retrieval text and sequence number.
      \item \textbf{External stub}: referenced entity outside the corpus.
    \end{itemize}
  \column{0.55\textwidth}
    \textbf{Edge types}
    \begin{itemize}
      \item Containment: document $\rightarrow$ provision $\rightarrow$ chunk.
      \item Reading order: \texttt{CHUNK\_NEXT}, \texttt{PROVISION\_NEXT}.
      \item Legal dependencies: \texttt{AMENDS}, \texttt{REPLACES}, \texttt{CITES}.
      \item Only verified directions may be traversed.
    \end{itemize}
\end{columns}
\vspace{0.15cm}
\begin{alertblock}{Integrity gating}
Quarantined records and unverified edge directions are excluded from trusted traversal to reduce graph poisoning.
\end{alertblock}
\end{frame}

\begin{frame}{500-Query Legal QA Benchmark}
\small
\begin{columns}[T]
  \column{0.48\textwidth}
    \begin{block}{Answerability}
      \begin{itemize}
        \item 400 answerable questions
        \item 100 unanswerable questions
        \item Unanswerable cases test hallucination resistance and abstention.
      \end{itemize}
    \end{block}
    \textbf{Answer types}
    \begin{itemize}
      \item Abstractive
      \item Boolean
      \item Extractive
      \item Unanswerable
    \end{itemize}
  \column{0.49\textwidth}
    \begin{block}{Question categories}
      \begin{itemize}
        \item Citation
        \item Cross-document
        \item Legal validity
        \item Multi-hop
        \item Single-hop
      \end{itemize}
    \end{block}
    \textbf{Difficulty levels}: easy, medium, and hard.
\end{columns}
\end{frame}

\begin{frame}{Ground Truth and Evaluation Contract}
\small
For each answerable question, the benchmark stores:
\[
Q_i = \{\mathrm{qa\_id},\ \mathrm{question},\ \mathrm{category},\ \mathrm{difficulty},
\ \mathrm{answer\_type},\ \mathrm{ref\_answer},\ \mathrm{gold\_chunks}\}
\]
\begin{columns}[T]
  \column{0.48\textwidth}
    \begin{block}{Retrieval metrics}
      \begin{itemize}
        \item Recall@$k$ and Hit@$k$
        \item MRR@$k$ and nDCG@$k$
        \item Precision@$k$ / chunk overlap
      \end{itemize}
    \end{block}
  \column{0.48\textwidth}
    \begin{block}{Generation metrics}
      \begin{itemize}
        \item Exact Match
        \item Token F1 and ROUGE-L
        \item Unanswerable accuracy
        \item Citation grounding and faithfulness
      \end{itemize}
    \end{block}
\end{columns}
\vspace{0.15cm}
\centering
\hi{Gold chunk IDs connect each benchmark answer directly to retrievable legal evidence.}
\end{frame}
