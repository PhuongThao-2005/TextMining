\vspace{-0.02cm}
\begin{block}{Evaluation contract}
Each query stores a reference answer and gold document/provision/chunk IDs. Retrieval uses Recall, Hit, MRR, nDCG, and Precision; generation uses Exact Match, F1/ROUGE-L, answerability, and citation coverage.
\end{block}
\end{frame}

\begin{frame}{Evaluation Metrics: Quality, Evidence \& Cost}
\tiny
\setlength{\itemsep}{0.02cm}
\begin{columns}[T,onlytextwidth]
  \column{0.48\textwidth}
    \begin{block}{Primary answer metrics}
      \begin{itemize}
        \item \textbf{Exact Match (EM):} normalized prediction equals normalized reference:
        \(\mathrm{EM}_i=\mathbf{1}[y_i=r_i]\).
        \item \textbf{Token F1:} with multiset overlap \(O\),
        \(P=O/L_y\), \(R=O/L_r\), and
        \(F_1=2PR/(P+R)\).
        \item \textbf{ROUGE-L:} the same F-score using the token-level longest
        common subsequence (LCS).
        \item \textbf{Paired delta:}
        \(\Delta_i(m)=m_{\mathrm{CoT},i}-m_{\mathrm{Base},i}\);
        averaged over shared valid \texttt{qa\_id}s with a paired-bootstrap
        95\% CI.
      \end{itemize}
    \end{block}
    \vspace{0.04cm}
    \emph{Scope:} lexical metrics use successful answerable outputs; paired
    statistics use only IDs valid in both runs.

  \column{0.49\textwidth}
    \begin{block}{Additional diagnostics}
      \begin{itemize}
        \item \textbf{Context Recall@$k$:}
        \(\lvert G\cap R_k\rvert/\lvert G\rvert\), the fraction of gold chunks
        retrieved.
        \item \textbf{Decision accuracy:} correct answer/refuse decisions
        divided by scheduled rows (500) or valid rows.
        \item \textbf{Citation presence:} successful answerable answers with a
        marker divided by successful answerable answers.
        \item \textbf{Structural coverage:} mean share of scored factual
        sentences carrying a marker.
        \item \textbf{Invalid markers:} invalid markers divided by all markers.
        \item \textbf{Completion:} successful rows / scheduled rows.
        \textbf{Latency:} mean, p50, and p95; milliseconds / 1000 = seconds.
      \end{itemize}
    \end{block}
\end{columns}
\vspace{0.04cm}
\begin{alertblock}{Interpretation boundary}
These are automatic overlap, abstention, citation-structure, and efficiency
diagnostics---not human judgments of legal correctness, citation entailment, or
reasoning quality.
\end{alertblock}
\end{frame}
