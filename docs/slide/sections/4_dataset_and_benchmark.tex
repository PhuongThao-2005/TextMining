\begin{frame}{Knowledge Graph Schema}
\small
\begin{columns}[T]
  \column{0.42\textwidth}
    \textbf{Node types}
    \begin{itemize}
      \item \textbf{Document}: authority, issuer, effective date, validity.
      \item \textbf{Provision}: article/clause header and unit type.
      \item \textbf{Chunk}: retrieval text and sequence number.
      \item \textbf{External stub}: referenced entity outside the corpus.
    \end{itemize}
  \column{0.55\textwidth}
    \textbf{Edge types}
    \begin{itemize}
      \item Containment: document $\rightarrow$ provision $\rightarrow$ chunk.
      \item Reading order: \texttt{CHUNK\_NEXT}, \texttt{PROVISION\_NEXT}.
      \item Legal dependencies: \texttt{AMENDS}, \texttt{REPLACES}, \texttt{CITES}.
      \item Only verified directions may be traversed.
    \end{itemize}
\end{columns}
\vspace{0.15cm}
\begin{alertblock}{Integrity gating}
Quarantined records and unverified edge directions are excluded from trusted traversal to reduce graph poisoning.
\end{alertblock}
\end{frame}

\begin{frame}{500-Query Legal QA Benchmark}
\small
\begin{columns}[T]
  \column{0.48\textwidth}
    \begin{block}{Answerability}
      \begin{itemize}
        \item 400 answerable questions
        \item 100 unanswerable questions
        \item Unanswerable cases test abstention under insufficient evidence.
      \end{itemize}
    \end{block}
    \textbf{Answer types}
    \begin{itemize}
      \item Abstractive
      \item Boolean
      \item Extractive
      \item Unanswerable
    \end{itemize}
  \column{0.49\textwidth}
    \begin{block}{Question categories}
      \begin{itemize}
        \item Single-hop: 265
        \item Citation: 106
        \item Legal validity: 96
        \item Multi-hop: 21
        \item Cross-document: 12
      \end{itemize}
    \end{block}

    \textbf{Generation model:} \texttt{gemini-3.1-flash-lite}

\end{columns}
\end{frame}

\begin{frame}{Ground Truth and Evaluation Contract}
\small

Each benchmark query stores metadata, a reference answer, and evidence provenance:

\[
Q_i =
\{
\texttt{qa\_id},
\texttt{question},
\texttt{category},
\texttt{answer\_type},
\texttt{reference\_answer},
\texttt{ground\_truth}
\}
\]

\[
G_i =
\{
\texttt{document\_ids},
\texttt{provision\_ids},
\texttt{chunk\_ids}
\}
\]
\vspace{-0.5em}
\begin{columns}[T]
  \column{0.48\textwidth}
    \begin{block}{Retrieval metrics}
      \begin{itemize}
        \item Recall@$k$ and Hit@$k$
        \item MRR@$k$ and nDCG@$k$
        \item Precision@$k$
      \end{itemize}
    \end{block}
  \column{0.48\textwidth}

    \begin{block}{Generation diagnostics}
      \begin{itemize}
        \item Exact Match
        \item Token F1 and ROUGE-L
        \item Answerability decision
        \item Citation presence / source coverage
      \end{itemize}
    \end{block}
\end{columns}
\vspace{0.15cm}
\centering
\hi{Retrieval metrics use gold chunk IDs on the 400 answerable questions; unanswerable questions have empty evidence sets.}

\end{frame}
