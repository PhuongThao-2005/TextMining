\section{Ablation Studies \& Results}

\begin{frame}{Controlled Experimental Protocol}
\scriptsize
\begin{columns}[T]
  \column{0.48\textwidth}
    \textbf{Shared settings}
    \begin{itemize}
      \item Frozen benchmark: 500 QA pairs.
      \item Retrieval metrics: 400 answerable cases.
      \item Dense encoder: \texttt{multilingual-e5-large}.
      \item Non-reasoning runs: \texttt{gpt-4o-mini}.
      \item Reasoning primary pair: fixed context, Base vs. CoT.
      \item Temperature 0; maximum 1,024 tokens.
      \item Comparisons paired by \texttt{qa\_id}.
    \end{itemize}
  \column{0.49\textwidth}
    \textbf{Uncertainty analysis}
    \begin{itemize}
      \item Mean per-question deltas.
      \item Percentile-bootstrap 95\% confidence intervals.
      \item 10,000 resamples for primary comparisons.
      \item A CI containing zero is treated as inconclusive.
    \end{itemize}
\end{columns}
\vspace{0.2cm}
\begin{alertblock}{Important}
Candidate budgets differ across experiment families, so scores are compared \textbf{within a family}, not as one pooled factorial study.
\end{alertblock}
\end{frame}

    \begin{itemize}

  \column{0.49\textwidth}
    \textbf{Graph effect}
    \begin{itemize}
      \item Recall: $-12.40$ pp (dense), $-11.81$ pp (hybrid).

      \item Expansion removes more gold evidence than it adds.
    \end{itemize}

\begin{frame}{Why Jina Is the Final Reranker Candidate}
\scriptsize
\begin{columns}[T]
  \column{0.48\textwidth}
    \begin{block}{Paired gain vs. RRF-only}
      \begin{itemize}
        \item Recall@10: \textbf{+9.23 pp}\\CI $[+5.51,+12.98]$
        \item MRR@10: \textbf{+13.35 pp}\\CI $[+9.51,+17.24]$
        \item nDCG@10: \textbf{+11.85 pp}\\CI $[+8.46,+15.27]$
      \end{itemize}
    \end{block}
  \column{0.49\textwidth}
    \begin{block}{Largest category gains}
      \begin{itemize}
        \item Legal validity ($n=91$): Recall \textbf{+14.47 pp}
        \item Citation ($n=106$): Recall \textbf{+11.89 pp}
        \item Abstractive ($n=199$): Recall \textbf{+13.65 pp}
        \item Boolean ($n=123$): Recall \textbf{+7.86 pp}
      \end{itemize}
    \end{block}
\end{columns}
\vspace{0.25cm}
\begin{alertblock}{Evidence ceiling}
Jina retains 88.50\% of gold chunks available in the shared candidate pool, compared with 71.29\% for RRF-only. The gain is better promotion, not a larger pool.
\end{alertblock}
\end{frame}

\begin{frame}{Primary Pair: What GPT-4o-mini CoT Changes}
\scriptsize
\begin{columns}[T]
  \column{0.49\textwidth}
    \textbf{Paired quality and latency}
    \begin{itemize}
      \item EM: $-0.25$ pp; CI $[-1.50,+1.00]$.
      \item Token F1: $-1.57$ pp; CI $[-2.24,-0.92]$.
      \item ROUGE-L: $-1.71$ pp; CI $[-2.34,-1.10]$.
      \item Token-F1 win/tie/loss: 79/164/157.
      \item Generation p50: 2.92 $\rightarrow$ 3.68 s.
      \item Total p50: 13.99 $\rightarrow$ 14.74 s.
    \end{itemize}
  \column{0.48\textwidth}
    \textbf{Behavior and evidence diagnostics}
    \begin{itemize}
      \item False refusal: 21.25\% $\rightarrow$ 19.50\%.
      \item Unanswerable recognition: 98.00\% for both.
      \item Citation presence: 78.25\% $\rightarrow$ 81.00\%.
      \item Structural coverage: 89.42\% $\rightarrow$ 87.91\%.
      \item Token F1: full $-1.98$ pp; partial $-1.86$ pp; no gold support $-1.30$ pp.
    \end{itemize}
\end{columns}
\vspace{0.15cm}
\begin{alertblock}{Interpretation boundary}
CoT lowers lexical overlap in every evidence stratum and adds latency. Decision and citation-marker diagnostics do not measure legal correctness, faithfulness, or citation entailment.
\end{alertblock}
\end{frame}

\begin{frame}{Efficiency and What the Metrics Do Not Prove}
\small
\begin{columns}[T]
  \column{0.48\textwidth}
    \textbf{Measured efficiency}
    \begin{itemize}
      \item Dense total p50: 5.483 s.
      \item Hybrid total p50: 31.538 s.
      \item Reranker p50: mMiniLM 0.385 s; Jina 1.278 s; BGE 2.872 s; Qwen3 9.695 s.
      \item Cached BM25 makes reranker latency incomplete for full-online deployment.
    \end{itemize}
  \column{0.49\textwidth}
    \textbf{Missing validation}
    \begin{itemize}
      \item Human-rated legal correctness and completeness.
      \item Claim faithfulness to retrieved passages.
      \item Citation precision and citation recall.
      \item Full-online p95 latency, throughput, and GPU memory.
    \end{itemize}
\end{columns}
\vspace{0.2cm}
\begin{alertblock}{Interpretation boundary}
Token F1, ROUGE-L, and refusal markers are diagnostics; they cannot certify that an answer is legally correct.
\end{alertblock}
\end{frame}
