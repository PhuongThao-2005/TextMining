\begin{frame}{Five-Stage Online Pipeline}
\small
\centering
\resizebox{0.98\linewidth}{!}{
\begin{tikzpicture}[
  node distance=0.35cm,
  box/.style={draw, rounded corners, align=center, minimum width=2.25cm, minimum height=1.0cm, fill=blue!7},
  graph/.style={box, fill=green!10},
  answer/.style={box, fill=orange!12},
  arrow/.style={-{Latex}, thick, UniBlue}
]
\node[box] (q) {User query\\+ date/scope};
\node[box, right=of q] (ret) {Dense E5\\+ BM25};
\node[graph, right=of ret] (kg) {Graph overlay\\+ expansion};
\node[box, right=of kg] (rank) {Fusion\\+ reranking};
\node[answer, right=of rank] (gen) {Grounded LLM\\+ citations};
\draw[arrow] (q)--(ret);
\draw[arrow] (ret)--(kg);
\draw[arrow] (kg)--(rank);
\draw[arrow] (rank)--(gen);
\end{tikzpicture}}
\vspace{0.4cm}
\begin{enumerate}
  \item Retrieve a broad candidate pool from complementary dense and sparse indexes.
  \item Apply legal whitelists and expand connected structural context.
  \item Fuse and rerank candidates, then send the top 10 chunks to the generator.
\end{enumerate}
\end{frame}

\begin{frame}{Dense and Sparse Retrieval}
\small
\begin{columns}[T]
  \column{0.49\textwidth}
    \begin{block}{Dense vector search}
      \begin{itemize}
        \item Model: \texttt{multilingual-e5-large} (560M, 1024 dimensions).
        \item FAISS \texttt{IndexFlatIP} over L2-normalized vectors.
        \item Embedding input combines document title, provision header, and chunk content.
        \item Strength: semantic paraphrases and natural-language questions.
      \end{itemize}
    \end{block}
  \column{0.49\textwidth}
    \begin{block}{Sparse keyword search}
      \begin{itemize}
        \item BM25 with $k_1=1.5$ and $b=0.75$.
        \item Vietnamese tokenization using \texttt{underthesea}.
        \item Strength: exact legal terms, document numbers, and citations.
        \item Dense and sparse lists are fused using RRF ($k=60$).
      \end{itemize}
    \end{block}
\end{columns}
\end{frame}

\begin{frame}{Knowledge Graph Operations}
\small
\begin{columns}[T]
  \column{0.49\textwidth}
    \textbf{Query-time reasoning}
    \begin{itemize}
      \item Fold validity events relative to \texttt{as\_of\_date}.
      \item Resolve conflicts by legal authority and version.
      \item Produce \texttt{id\_str\_filter} whitelists for retrieval.
      \item Traverse only direction-verified legal edges.
    \end{itemize}
  \column{0.49\textwidth}
    \textbf{Structural expansion}
    \begin{itemize}
      \item \texttt{DOCUMENT\_HAS\_PROVISION}
      \item \texttt{PROVISION\_HAS\_CHUNK}
      \item \texttt{CHUNK\_NEXT}
      \item \texttt{PROVISION\_NEXT}
    \end{itemize}
    \vspace{0.1cm}
    Sibling chunks are expanded in reading order to restore missing local context.
\end{columns}
\vspace{0.2cm}
\begin{block}{Implementation status}
The in-memory loader, builder, traversal, expansion, overlay, and context modules pass all 27 knowledge-graph tests.
\end{block}
\end{frame}

\begin{frame}{Sequential Multi-Reranker Stage}
\small
\begin{columns}[T]
  \column{0.54\textwidth}
    \textbf{Supported backends}
    \begin{enumerate}
      \item mMiniLMv2 (117M): lightweight baseline.
      \item BGE-Reranker-v2-m3 (567M): multilingual reranker.
      \item Jina-Reranker-v2 (278M): multilingual long-context model.
      \item Qwen3-Reranker-0.6B: causal-LM reranker in FP16.
    \end{enumerate}
  \column{0.43\textwidth}
    \textbf{GPU-safe execution}
    \[
      \mathrm{Load}(M_i)
      \rightarrow \mathrm{Run}(M_i)
    \]
    \[
      \rightarrow \mathrm{Unload}(M_i)
      \rightarrow \mathrm{empty\_cache}()
    \]
    Candidates: $K=30$\\
    Final evidence: $N=10$
\end{columns}
\vspace{0.2cm}
\begin{alertblock}{Why rerank?}
Unlike separate bi-encoder vectors, a cross-encoder jointly attends to the complete query and candidate passage.
\end{alertblock}
\end{frame}

\begin{frame}{Citation-Grounded Answer Generation}
\small
\begin{columns}[T]
  \column{0.48\textwidth}
    \begin{block}{Generator configuration}
      \begin{itemize}
        \item Model: \texttt{gpt-4o-mini}
        \item Temperature: 0.0
        \item Maximum output: 1,024 tokens
        \item Context: top 10 reranked chunks
      \end{itemize}
    \end{block}
  \column{0.49\textwidth}
    \begin{block}{Prompt contract}
      \begin{itemize}
        \item Cite governing legal provisions explicitly.
        \item Use only retrieved evidence.
        \item Return \texttt{INSUFFICIENT\_CONTEXT} when evidence is inadequate.
        \item Format boolean, extractive, abstractive, and unanswerable outputs appropriately.
      \end{itemize}
    \end{block}
\end{columns}
\vspace{0.25cm}
\centering
\hi{The LLM synthesizes evidence; it is not treated as the legal source.}
\end{frame}
